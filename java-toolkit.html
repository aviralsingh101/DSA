Fast IO and longs

JavaFastIO.java
import java.io.*;
import java.util.*;

public class FastIO {
    static StreamTokenizer st = new StreamTokenizer(
        new BufferedReader(new InputStreamReader(System.in)));
    static int ni() throws IOException { st.nextToken(); return (int) st.nval; }
    static long nl() throws IOException { st.nextToken(); return (long) st.nval; }

    static long gcd(long a, long b) { return b == 0 ? a : gcd(b, a % b); }
    static long lcm(long a, long b) { return a / gcd(a, b) * b; } // divide first

    static final int MOD = 1_000_000_007;
    static long modPow(long a, long e) {
        long r = 1; a %= MOD;
        while (e > 0) {
            if ((e & 1) == 1) r = r * a % MOD;
            a = a * a % MOD; e >>= 1;
        }
        return r;
    }
    static long modInv(long a) { return modPow(a, MOD - 2); } // MOD prime
}

Binary search firstTrue

JavaFirstTrue.java
static int firstTrue(int lo, int hi, java.util.function.IntPredicate pred) {
    while (lo < hi) {
        int mid = lo + (hi - lo) / 2;
        if (pred.test(mid)) hi = mid;
        else lo = mid + 1;
    }
    return lo;
}

DSU

JavaDSU.java
static class DSU {
    final int[] p, sz;
    int comps;
    DSU(int n) {
        p = new int[n]; sz = new int[n]; comps = n;
        for (int i = 0; i < n; i++) { p[i] = i; sz[i] = 1; }
    }
    int find(int x) { return p[x] == x ? x : (p[x] = find(p[x])); }
    boolean union(int a, int b) {
        a = find(a); b = find(b);
        if (a == b) return false;
        if (sz[a] < sz[b]) { int t = a; a = b; b = t; }
        p[b] = a; sz[a] += sz[b]; comps--;
        return true;
    }
}

Fenwick (BIT)

JavaFenwick.java
static class Fenwick {
    final long[] t;
    Fenwick(int n) { t = new long[n + 1]; }
    void add(int i, long v) { for (i++; i < t.length; i += i & -i) t[i] += v; }
    long sum(int i) {              // sum of [0, i]
        long s = 0; for (i++; i > 0; i -= i & -i) s += t[i]; return s;
    }
    long range(int l, int r) { return sum(r) - sum(l - 1); }
}

Segment tree (point update, range min)

JavaSegTree.java
static class SegTree {
    final int n; final int[] t;
    SegTree(int[] a) {
        n = a.length; t = new int[4 * n];
        build(1, 0, n - 1, a);
    }
    void build(int v, int l, int r, int[] a) {
        if (l == r) { t[v] = a[l]; return; }
        int m = l + (r - l) / 2;
        build(v * 2, l, m, a); build(v * 2 + 1, m + 1, r, a);
        t[v] = Math.min(t[v * 2], t[v * 2 + 1]);
    }
    void upd(int v, int l, int r, int i, int x) {
        if (l == r) { t[v] = x; return; }
        int m = l + (r - l) / 2;
        if (i <= m) upd(v * 2, l, m, i, x); else upd(v * 2 + 1, m + 1, r, i, x);
        t[v] = Math.min(t[v * 2], t[v * 2 + 1]);
    }
    int qry(int v, int l, int r, int ql, int qr) {
        if (qr < l || r < ql) return Integer.MAX_VALUE;
        if (ql <= l && r <= qr) return t[v];
        int m = l + (r - l) / 2;
        return Math.min(qry(v * 2, l, m, ql, qr), qry(v * 2 + 1, m + 1, r, ql, qr));
    }
}

Dijkstra

JavaDijkstra.java
static long[] dijkstra(List<List<int[]>> g, int src) {
    int n = g.size();
    long[] d = new long[n];
    Arrays.fill(d, Long.MAX_VALUE / 4);
    d[src] = 0;
    PriorityQueue<long[]> pq = new PriorityQueue<>(Comparator.comparingLong(a -> a[0]));
    pq.offer(new long[] {0, src});
    while (!pq.isEmpty()) {
        long[] cur = pq.poll();
        int u = (int) cur[1];
        if (cur[0] != d[u]) continue;
        for (int[] e : g.get(u)) {
            int v = e[0], w = e[1];
            if (d[v] > d[u] + w) {
                d[v] = d[u] + w;
                pq.offer(new long[] {d[v], v});
            }
        }
    }
    return d;
}

Binary-lifting LCA

JavaLCA.java
static class LCA {
    final int LOG; final int[][] up; final int[] depth;
    LCA(List<List<Integer>> g, int root) {
        int n = g.size(); LOG = 32 - Integer.numberOfLeadingZeros(n);
        up = new int[LOG][n]; depth = new int[n];
        dfs(g, root, root);
        for (int k = 1; k < LOG; k++)
            for (int v = 0; v < n; v++) up[k][v] = up[k - 1][up[k - 1][v]];
    }
    void dfs(List<List<Integer>> g, int v, int p) {
        up[0][v] = p;
        for (int to : g.get(v)) if (to != p) {
            depth[to] = depth[v] + 1; dfs(g, to, v);
        }
    }
    int lift(int v, int d) {
        for (int k = 0; d > 0; k++, d >>= 1) if ((d & 1) == 1) v = up[k][v];
        return v;
    }
    int lca(int a, int b) {
        if (depth[a] < depth[b]) { int t = a; a = b; b = t; }
        a = lift(a, depth[a] - depth[b]);
        if (a == b) return a;
        for (int k = LOG - 1; k >= 0; k--)
            if (up[k][a] != up[k][b]) { a = up[k][a]; b = up[k][b]; }
        return up[0][a];
    }
}

KMP prefix function

JavaKMP.java
static int[] prefixFunction(String s) {
    int n = s.length(); int[] pi = new int[n];
    for (int i = 1; i < n; i++) {
        int j = pi[i - 1];
        while (j > 0 && s.charAt(i) != s.charAt(j)) j = pi[j - 1];
        if (s.charAt(i) == s.charAt(j)) j++;
        pi[i] = j;
    }
    return pi;
}

Sieve + SPF

JavaSieve.java
static int[] spf(int n) {
    int[] s = new int[n + 1];
    for (int i = 2; i <= n; i++) {
        if (s[i] == 0) {
            s[i] = i;
            if ((long) i * i <= n) for (int j = i * i; j <= n; j += i)
                if (s[j] == 0) s[j] = i;
        }
    }
    return s;
}
